<div className="bg-white rounded-2xl shadow-[0_1px_2px_0_rgba(15,23,42,0.04)] border border-ink-100">

          <div className="flex items-center gap-1">

              <ChevronRight className="w-4 h-4" strokeWidth={2} />
            </button>
          </div>
        </div>

        <div className="sm:hidden border-b border-ink-100 overflow-x-auto no-scrollbar">
          <div className="flex px-2 py-2 gap-1 min-w-max">
            {weekDays.map((day, i) => (
              <button
                key={i}
                onClick={() => !day.past && setSelectedDayIndex(i)}
                disabled={day.past}
                className={`flex flex-col items-center px-3 py-2 rounded-xl shrink-0 transition cursor-pointer ${
                  day.past
                    ? 'opacity-30 cursor-not-allowed'
                    : selectedDayIndex === i
                    ? 'bg-brand-600 text-white'
                    : day.today
                    ? 'bg-brand-50 border border-brand-200 text-brand-700'
                    : 'hover:bg-ink-50 text-ink-700'
                }`}
              >
                <span className="text-[10px] font-semibold">{day.short}</span>
                <span className="text-sm font-bold">{day.date}</span>
              </button>
            ))}
          </div>
        </div>

        {/* MOBİL: Günlük slot listesi */}
        <div className="sm:hidden divide-y divide-ink-50">
          {hours.map((hour) => {
            const key = `${hour}-${selectedDayIndex}`
            const status = specSlots[key] as SlotType | undefined
            const day = weekDays[selectedDayIndex]
            return (
              <div key={hour} className="flex items-center gap-3 px-4 py-2.5">
                <span className="text-xs text-ink-400 font-medium w-12 shrink-0">{hour}</span>
                {status === 'closed' ? (
                  <div className="flex-1 h-9 rounded-xl border border-ink-100"
                    style={{ backgroundImage: 'repeating-linear-gradient(45deg,#f8fafc,#f8fafc 4px,#f1f5f9 4px,#f1f5f9 8px)' }} />
                ) : status === 'taken' ? (
                  <div className="flex-1 h-9 rounded-xl bg-ink-100 border border-ink-200" />
                ) : (
                  <button
                    onClick={() => handleSlotClick(hour, selectedDayIndex, day)}
                    className="flex-1 h-9 rounded-xl bg-emerald-50 border border-dashed border-emerald-300 text-emerald-700 text-xs font-semibold cursor-pointer hover:bg-emerald-100 transition"
                  >
                    Müsait
                  </button>
                )}
              </div>
            )
          })}
        </div>

        {/* MASAÜSTÜ: Haftalık tablo */}
        <div className="hidden sm:block overflow-x-auto">
          <div className="min-w-[560px]">
            <div className="grid grid-cols-[60px_repeat(7,minmax(0,1fr))] bg-ink-50/60 border-b border-ink-100">
              <div className="border-r border-ink-100" />
              {weekDays.map((day, i) => (
                <div key={i} className={`px-2 py-2.5 text-xs font-semibold text-center border-r border-ink-100 last:border-r-0 ${day.today ? 'bg-brand-50' : day.past ? 'opacity-40' : ''}`}>
                  <div className={day.today ? 'text-brand-600' : 'text-ink-500'}>{day.short}</div>
                  <div className={`text-sm font-bold ${day.today ? 'text-brand-800' : 'text-ink-800'}`}>{day.date}</div>
                </div>
              ))}
            </div>
            {hours.map((hour) => (
              <div key={hour} className="grid grid-cols-[60px_repeat(7,minmax(0,1fr))] border-b border-ink-50">
                <div className="px-2 py-2 text-[11px] text-ink-400 font-medium border-r border-ink-100 flex items-center justify-end pr-3">{hour}</div>
                {weekDays.map((day, dayIndex) => {
                  const key = `${hour}-${dayIndex}`
                  const status = specSlots[key] as SlotType | undefined
                  if (day.past) return (
                    <div key={dayIndex} className="p-1 border-r border-ink-50 last:border-r-0">
                      <div className="h-9 rounded-md bg-ink-50 opacity-30" />
                    </div>
                  )
                  if (status === 'closed') return (
                    <div key={dayIndex} className="p-1 border-r border-ink-50 last:border-r-0">
                      <div className="h-9 rounded-md border border-ink-100"
                        style={{ backgroundImage: 'repeating-linear-gradient(45deg,#f8fafc,#f8fafc 4px,#f1f5f9 4px,#f1f5f9 8px)' }} />
                    </div>
                  )
                  if (status === 'taken') return (
                    <div key={dayIndex} className="p-1 border-r border-ink-50 last:border-r-0">
                      <div className="h-9 rounded-md bg-ink-100 border border-ink-200" />
                    </div>
                  )
                  return (
                    <div key={dayIndex} className="p-1 border-r border-ink-50 last:border-r-0">
                      <div
                        onClick={() => handleSlotClick(hour, dayIndex, day)}
                        className={`h-9 rounded-md bg-emerald-50 border border-dashed border-emerald-300 cursor-pointer hover:bg-emerald-100 transition ${day.today ? 'ring-1 ring-emerald-400' : ''}`}
                      />
                    </div>
                  )
                })}
              </div>
            ))}
          </div>
        </div>

        {/* Footer */}
        <div className="px-4 sm:px-5 py-3 border-t border-ink-100 flex items-center gap-4 text-xs text-ink-500 flex-wrap">

            { color: '', label: 'Kapalı', striped: true },
          ].map((item) => (
            <span key={item.label} className="flex items-center gap-1.5">
              <span className={`w-3 h-3 rounded border ${item.color}`}
                style={item.striped ? { backgroundImage: 'repeating-linear-gradient(45deg,#f8fafc,#f8fafc 4px,#f1f5f9 4px,#f1f5f9 8px)', borderColor: '#e2e8f0' } : {}} />
              {item.label}
            </span>
          ))}
        </div>
